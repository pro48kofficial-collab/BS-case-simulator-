import random import time # Словник скінів з твоїх зображень (1000003947.png, 1000003912.jpg, 1000003913.jpg і т.д.) SKINS_DATA = { "Base": [ {"name": "AK-47 | Exclusion Zone", "price": 25}, ], "Rare": [ {"name": "M4A1 | Bubble Gum", "price": 60}, ], "Legendary": [ {"name": "Bat | XLoad", "price": 250}, {"name": "Kunai | Overflow", "price": 300}, {"name": "Bowie Knife | Neon Flake", "price": 350}, {"name": "Kunai | Cyber Rainbow", "price": 400}, {"name": "Bizon | Tesla Gold", "price": 220}, ], "Secret": [ {"name": "M4A1 | Dolphin Laser", "price": 900}, {"name": "Dual Daggers | Skull Puppets", "price": 1100}, {"name": "M9 Bayonet | Bio Hazard", "price": 1300}, ] } # Шанси випадіння (у відсотках) CHANCES = { "Base": 50, "Rare": 28, "Legendary": 16, "Secret": 6 } # Кольори для консолі COLORS = { "Base": "\033[90m", # Сірий "Rare": "\033[94m", # Синій "Legendary": "\033[95m", # Пурпурний "Secret": "\033[91m", # Червоний "Gold": "\033[93m", # Золотий "END": "\033[0m" # Скидання } class CaseSimulator: def __init__(self): self.balance = 300 self.case_price = 60 self.inventory = [] def spin_case(self): if self.balance < self.case_price: print(f"\n{COLORS['Secret']}❌ Недостатньо монет! Потрібно {self.case_price} бс-коінів.{COLORS['END']}") return self.balance -= self.case_price print("\n🎰 Відкриваємо кейс...") for _ in range(3): print(".", end="", flush=True) time.sleep(0.4) print("\n") rarity_choices = list(CHANCES.keys()) rarity_weights = list(CHANCES.values()) dropped_rarity = random.choices(rarity_choices, weights=rarity_weights, k=1)[0] dropped_skin = random.choice(SKINS_DATA[dropped_rarity]) self.inventory.append({ "name": dropped_skin["name"], "rarity": dropped_rarity, "price": dropped_skin["price"] }) color = COLORS[dropped_rarity] print(f"🎉 ТОБІ ВИПАЛО: {color}[{dropped_rarity}] {dropped_skin['name']}{COLORS['END']}!") print(f"Ціна скіна: {COLORS['Gold']}{dropped_skin['price']} монет{COLORS['END']}") def show_inventory(self): if not self.inventory: print(f"\n🎒 Твій інвентар порожній.") return print(f"\n🎒 Твій Інвентар ({len(self.inventory)} предметів):") for idx, item in enumerate(self.inventory, 1): color = COLORS[item['rarity']] print(f"{idx}. {color}[{item['rarity']}] {item['name']}{COLORS['END']} — {COLORS['Gold']}{item['price']} монет{COLORS['END']}") def sell_item(self): if not self.inventory: print("\n❌ Немає чого продавати.") return self.show_inventory() try: choice = int(input("\nВведіть номер предмета для продажу: ")) - 1 if 0 <= choice < len(self.inventory): item = self.inventory.pop(choice) self.balance += item['price'] print(f"\n✅ Продано: {item['name']} за {COLORS['Gold']}{item['price']} монет{COLORS['END']}.") else: print("\n❌ Невірний номер предмета.") except ValueError: print("\n❌ Будь ласка, введіть число.") def get_free_coins(self): reward = random.randint(40, 100) self.balance += reward print(f"\n🎁 Бонус! Ти отримав {COLORS['Gold']}{reward} безкоштовних монет{COLORS['END']}.") def play(self): print(f"{COLORS['Gold']}=== BLOCK STRIKE EXCLUSIVE CASE ==={COLORS['END']}") while True: print(f"\n💰 Баланс: {COLORS['Gold']}{self.balance} монет{COLORS['END']}") print(f"1. Відкрити Оновлений Кейс (Ціна: {self.case_price} монет)") print("2. Подивитися Інвентар") print("3. Продати скін") print("4. Отримати безкоштовні монети") print("5. Вийти з гри") choice = input("\nОбери дію (1-5): ").strip() if choice == "1": self.spin_case() elif choice == "2": self.show_inventory() elif choice == "3": self.sell_item() elif choice == "4": self.get_free_coins() elif choice == "5": print("\nДякуємо за гру! Удачі в БС!") break else: print("\n❌ Невідома команда.") if __name__ == "__main__": simulator = CaseSimulator() simulator.play()
